            function baixarArquivo(url, nomeArquivo) {

                if (exportInProgress) return;



                const filtro = montarFiltro(1);

                if (!filtro) {

                    return;

                }



                const spinner = $('#spinnerBusca');

                ocultarAlerta();

                alternarEstadoExportacao(true);

                spinner.removeClass('d-none');



                fetch(url, {

                    method: 'POST',

                    headers: { 'Content-Type': 'application/json' },

                    body: JSON.stringify(filtro)

                })

                    .then(async response => {

                        if (!response.ok) {

                            let mensagem = 'Falha ao exportar os dados.';

                            try {

                                const erro = await response.json();

                                mensagem = erro?.mensagem || erro?.message || mensagem;

                            } catch {

                                const texto = await response.text();

                                if (texto) mensagem = texto;

                            }



                            throw new Error(mensagem);

                        }



                        return response.blob();

                    })

                    .then(blob => {

                        const objectUrl = URL.createObjectURL(blob);

                        const a = document.createElement('a');

                        a.href = objectUrl;

                        a.download = nomeArquivo;

                        document.body.appendChild(a);

                        a.click();

                        a.remove();

                        URL.revokeObjectURL(objectUrl);

                    })

                    .catch(err => mostrarAlerta(err.message))

                    .finally(() => {

                        spinner.addClass('d-none');

                        alternarEstadoExportacao(false);

                    });

            }
